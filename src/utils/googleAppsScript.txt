function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🚀 KOC Dashboard')
    .addItem('✨ 1. Tự động tạo Bảng chuẩn & Dữ liệu mẫu', 'khoiTaoBangTinhKOC')
    .addItem('📋 2. Xem link kết nối Dashboard', 'layLinkKetNoi')
    .addToUi();
}

function layLinkKetNoi() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var url = ss.getUrl();
  var htmlOutput = HtmlService
    .createHtmlOutput(
      '<div style=\"font-family: sans-serif; padding: 15px; font-size: 13px;\">' +
      '<p><b>Link Google Sheet của bạn:</b></p>' +
      '<input style=\"width: 100%; padding: 8px; font-size: 12px;\" readonly value=\"' + url + '\" id=\"sheetUrl\">' +
      '<p style=\"color: #059669; font-weight: bold; margin-top: 15px;\">👉 Hướng dẫn kết nối:</p>' +
      '<ol style=\"line-height: 1.6; color: #334155;\">' +
      '<li>Sao chép link ở trên.</li>' +
      '<li>Bấm nút <b>Chia sẻ (Share)</b> ở góc trên bên phải -> Chọn <b>\"Bất kỳ ai có liên kết đều có thể xem\"</b>.</li>' +
      '<li>Quay lại <b>KOC Power Dashboard</b>, dán link vào và bấm <b>Lưu & Đồng bộ</b>.</li>' +
      '</ol>' +
      '</div>'
    )
    .setWidth(450)
    .setHeight(280);
  SpreadsheetApp.getUi().showModalDialog(htmlOutput, '🔗 Link kết nối Dashboard');
}
